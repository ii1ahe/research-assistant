% ========== SLIDE 2: PROBLEM =========================================
\begin{frame}{The problem in one sentence}
  \begin{block}{What the system does}
    You ask one research question; the system retrieves evidence from Wikipedia, arXiv and the
    open web \emph{concurrently}, and returns one answer whose numbered citations point at the
    sources it actually used.
  \end{block}

  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \textbf{Inputs}
  \begin{itemize}\setlength\itemsep{0.25em}
    \item a question (non-empty, $\le$500 chars)
    \item sources: \texttt{wikipedia}, \texttt{arxiv}, \texttt{web}
    \item a results-per-source limit (1--20, default 3)
  \end{itemize}

  \column{0.48\textwidth}
  \textbf{Outputs}
  \begin{itemize}\setlength\itemsep{0.25em}
    \item a cited answer + a numbered source list
    \item a per-source retrieval report
    \item a session row, and an exit status
  \end{itemize}
  \end{columns}

  \vspace{0.5em}
  \textbf{Entry points:} \texttt{python -m researcher ask} and \texttt{demo} (five supplied
  questions). \textbf{No HTTP API} --- ADR-006; the container publishes no port.

  \vspace{0.3em}
  {\footnotesize Exit status is part of the contract: \textbf{0} answered (possibly partial),
  \textbf{1} no usable answer, \textbf{2} bad input or configuration.}
\end{frame}

% ========== SLIDE 4: KEY DESIGN DECISIONS ============================
\begin{frame}{Three design decisions we can defend}
  \begin{enumerate}\setlength\itemsep{0.5em}
    \item \textbf{Protocols over abstract base classes, at both storage seams.}\\
      An ABC imposes a \emph{nominal} relationship, so a test double must inherit from the real
      implementation. A \texttt{Protocol} is structural --- which is why \texttt{memory.py} is a
      genuine second implementation with the \emph{same} contract suite run against both.
    \item \textbf{A semaphore, not a bigger \texttt{gather}.}\\
      The bound on in-flight sources is configuration (1--16), not the shape of the code. And
      each fetch carries its own deadline rather than one wrapping the group --- a group deadline
      lets one slow source destroy two fast results.
    \item \textbf{PostgreSQL for both the cache and the sessions} (ADR-002).\\
      One store to operate; \texttt{ON CONFLICT} makes the database enforce the upsert; and
      \texttt{PERSIST\_SESSIONS=false} skips session writes, not cache use.
  \end{enumerate}

  \vspace{0.15em}
  \begin{block}{Expect to be asked}
    ``Why not threads? Why not SQLite? Why not an HTTP API?'' --- answers in ADR-005, ADR-002,
    ADR-006.
  \end{block}
\end{frame}

% ========== SLIDE 5: CONCURRENCY (THE BENCHMARK) =====================
\begin{frame}{Concurrency: the benchmark}
  \textbf{Workload:} the five supplied sample questions, three sources, three results each, live
  APIs, no cache reuse (\texttt{python scripts/bench.py --pause 10}).

  \vspace{0.3em}
  \begin{center}\small
  \begin{tabular}{lccc}
    \toprule
    & \textbf{Sequential (b=1)} & \textbf{Concurrent (b=3)} & \textbf{Speedup} \\
    \midrule
    Retrieval (5 questions)   & 4.83\,s & 2.89\,s & \textbf{1.67$\times$} \\
    Retrieval (per question)  & 0.97\,s & 0.58\,s & \textbf{1.67$\times$} \\
    End to end (5 questions)  & 43.4\,s & 43.3\,s & \textbf{1.00$\times$} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.25em}
  \textbf{Bottleneck named:} synthesis is \emph{one sequential model call}, 89\,\% of wall-clock
  time. The concurrency design optimised the other 11\,\%.

  \vspace{0.2em}
  \textbf{What didn't reach $N\times$:} the ceiling is not the source count but the ratio of
  summed to slowest latency --- \textbf{1.83$\times$} here, so 1.67$\times$ is
  \textbf{92\,\% of the ceiling}.
\end{frame}

% ========== SLIDE 6: ROBUSTNESS STORY ================================
\begin{frame}{Robustness: one concrete failure mode}
  \textbf{Scenario:} mid-verification, Gemini returned \texttt{429 RESOURCE\_EXHAUSTED}
  (\texttt{limit: 20}, \texttt{Please retry in 21.9s}). Two of five demo questions unanswered.

  \vspace{0.2em}
  \textbf{How we surfaced it:} we suspected the new container and \emph{refuted} it --- the same
  question from the host answered in 2.92\,s. Then we probed the provider directly to see the raw
  payload behind our deliberately generic message.

  \vspace{0.2em}
  \textbf{What the system does:}
  \begin{itemize}\setlength\itemsep{0.15em}
    \item \textbf{Isolate:} \texttt{gather(return\_exceptions=True)} + a \texttt{SourceOutcome}
      per source --- one question lost, not the run.
    \item \textbf{Disclose:} the CLI names the failed source; exit 1 separates ``some questions
      failed'' from ``the program broke''.
    \item \textbf{Contain:} provider text never reaches a log --- it becomes an
      \texttt{UpstreamError} with a safe message.
  \end{itemize}

  \vspace{0.2em}
  \textbf{\textcolor{accent2}{The gap we are naming, not hiding:}} we wait a jittered
  $\le$0.5\,s against a \textbf{21.9\,s} hint, so all three attempts land in one window.
\end{frame}

% ========== SLIDE 9: LIMITATIONS + FUTURE ============================
\begin{frame}{Limitations \& what we'd do next}
  \textbf{Where the system is brittle today}
  \begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{Answer correctness is not verified.} A fluent, correctly-cited, \emph{wrong}
      answer passes every check we have --- we validate structure, never support.
    \item \textbf{Retry hints are honoured, not anticipated.} The provider's ``retry in 21.9s''
      message, and a \texttt{Retry-After} header when the message names no delay, are parsed and
      waited --- but the first call still discovers a spent quota by being refused.
    \item \textbf{Concurrency stops at the question boundary} --- the demo's five questions run
      sequentially, and \texttt{MAX\_PARALLEL\_SOURCES} bounds sources within a question only.
  \end{itemize}

  \vspace{0.35em}
  \textbf{Given another week we would}
  \begin{enumerate}\setlength\itemsep{0.25em}
    \item Add a token bucket per provider, so a quota is anticipated instead of discovered by
      the first refused call.
    \item Stream the synthesis, or overlap it with retrieval --- the \emph{only} change that
      makes the concurrency work visible end to end (1.67$\times$ $\to$ a real end-to-end win).
    \item Move PostgreSQL off \texttt{trust} auth to \texttt{scram-sha-256}, and add a grounding
      check between each claim and its cited source.
  \end{enumerate}
\end{frame}
